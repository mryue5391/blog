硬核解决音量异常变化。
编写了一个C程序，侦测音量大小，如果低于预设值，立刻拉回正常水平。

代码展示：
**声明：** 本代码仅供学习与技术交流使用，严禁用于任何非法、未经授权或恶意用途。

```C
#include <windows.h>
#include <mmdeviceapi.h>
#include <endpointvolume.h>
#include <stdio.h>
#include <stdbool.h>
#include <wchar.h>

static const GUID My_CLSID_MMDeviceEnumerator = {0xBCDE0395, 0xE52F, 0x467C, {0x8E, 0x3D, 0xC4, 0x57, 0x92, 0x91, 0x69, 0x2E}};
static const GUID My_IID_IMMDeviceEnumerator = {0xA95664D2, 0x9614, 0x4F35, {0xA7, 0x46, 0xDE, 0x8D, 0xB6, 0x36, 0x17, 0xE6}};
static const GUID My_IID_IAudioEndpointVolume = {0x5CDF2C82, 0x841E, 0x4546, {0x97, 0x22, 0x0C, 0xF7, 0x40, 0x78, 0x22, 0x9A}};

static const PROPERTYKEY My_PKEY_Device_FormFactor = {{0xA45C254E, 0xDF1C, 0x4EFD, {0x80, 0x20, 0x67, 0xD1, 0x46, 0xA8, 0x50, 0xE0}}, 0};
static const PROPERTYKEY My_PKEY_Device_FriendlyName = {{0xA45C254E, 0xDF1C, 0x4EFD, {0x80, 0x20, 0x67, 0xD1, 0x46, 0xA8, 0x50, 0xE0}}, 14};

bool CheckIfDeviceIsHeadphone(IMMDevice* pDevice) {
    IPropertyStore* pProps = NULL;
    HRESULT hr = pDevice->lpVtbl->OpenPropertyStore(pDevice, STGM_READ, &pProps);
    if (FAILED(hr)) return false;

    bool isHeadphone = false;

    PROPVARIANT varForm;
    PropVariantInit(&varForm);
    hr = pProps->lpVtbl->GetValue(pProps, &My_PKEY_Device_FormFactor, &varForm);
    if (SUCCEEDED(hr) && varForm.vt == VT_UI4) {
        if (varForm.ulVal == 2 || varForm.ulVal == 3) {
            isHeadphone = true;
        }
    }
    PropVariantClear(&varForm);

    if (!isHeadphone) {
        PROPVARIANT varName;
        PropVariantInit(&varName);
        hr = pProps->lpVtbl->GetValue(pProps, &My_PKEY_Device_FriendlyName, &varName);
        if (SUCCEEDED(hr) && varName.vt == VT_LPWSTR && varName.pwszVal != NULL) {
            if (wcsstr(varName.pwszVal, L"Head") != NULL ||
                wcsstr(varName.pwszVal, L"head") != NULL ||
                wcsstr(varName.pwszVal, L"Ear") != NULL ||
                wcsstr(varName.pwszVal, L"ear") != NULL ||
                wcsstr(varName.pwszVal, L"耳") != NULL) {
                isHeadphone = true;
            }
        }
        PropVariantClear(&varName);
    }

    pProps->lpVtbl->Release(pProps);
    return isHeadphone;
}

int main() {
    CoInitialize(NULL);

    IMMDeviceEnumerator* pEnumerator = NULL;
    IAudioEndpointVolume* pVolume = NULL;
    IMMDevice* pDevice = NULL;

    HRESULT hr = CoCreateInstance(&My_CLSID_MMDeviceEnumerator, NULL, CLSCTX_INPROC_SERVER, 
                                  &My_IID_IMMDeviceEnumerator, (void**)&pEnumerator);
    if (FAILED(hr)) {
        CoUninitialize();
        return 1;
    }

    bool wasHeadphone = false;
    bool firstRun = true;

    while (true) {
        hr = pEnumerator->lpVtbl->GetDefaultAudioEndpoint(pEnumerator, eRender, eConsole, &pDevice);
        
        if (SUCCEEDED(hr)) {
            bool isHeadphone = CheckIfDeviceIsHeadphone(pDevice);
            
            hr = pDevice->lpVtbl->Activate(pDevice, &My_IID_IAudioEndpointVolume, CLSCTX_INPROC_SERVER, NULL, (void**)&pVolume);
            
            if (SUCCEEDED(hr)) {
                if (isHeadphone) {
                    wasHeadphone = true;
                    
                    float currentVolume = 0.0f;
                    pVolume->lpVtbl->GetMasterVolumeLevelScalar(pVolume, &currentVolume);
                    int volPercent = (int)(currentVolume * 100.0f + 0.5f); 

                    if (volPercent < 10) {
                        pVolume->lpVtbl->SetMasterVolumeLevelScalar(pVolume, 0.10f, NULL);
                    }

                    BOOL isMuted = FALSE;
                    pVolume->lpVtbl->GetMute(pVolume, &isMuted);
                    if (isMuted) {
                        pVolume->lpVtbl->SetMute(pVolume, FALSE, NULL);
                    }
                } 
                else {
                    if (wasHeadphone || firstRun) {
                        pVolume->lpVtbl->SetMasterVolumeLevelScalar(pVolume, 0.0f, NULL);
                        wasHeadphone = false;
                        firstRun = false;
                    }
                }
                pVolume->lpVtbl->Release(pVolume);
            }
            pDevice->lpVtbl->Release(pDevice);
        }

        Sleep(1);
    }

    pEnumerator->lpVtbl->Release(pEnumerator);
    CoUninitialize();
    return 0;
}
```
